```python
def chat_stream(model, messages, temperature=0.7, num_ctx=16384, timeout=900):
    """Generator over /api/chat with stream=True: yields text deltas.

    Returns (via StopIteration.value) the same metrics dict as chat(). The
    default temperature is conversational, not the 0.1 agents use: this feeds
    the Studio chat tab, not a persona.
    """
    payload = {
        "model": model,
        "messages": messages,
        "stream": True,
        "options": {"temperature": temperature, "num_ctx": num_ctx},
    }
    req = urllib.request.Request(
        OLLAMA_URL, data=json.dumps(payload).encode("utf-8"),
        headers={"Content-Type": "application/json"},
    )
    t0 = time.time()
    parts = []
    final = {}
    with urllib.request.urlopen(req, timeout=timeout) as resp:
        for line in resp:
            line = line.strip()
            if not line:
                continue
            body = json.loads(line.decode("utf-8"))
            text = body.get("message", {}).get("content", "")
            if text:
                parts.append(text)
                yield text
            if body.get("done"):
                final = body
    metrics = _metrics(final, time.time() - t0)
    metrics["content"] = "".join(parts)
    return metrics
```

```python
"""Chat sessions on disk: one session, one JSON file.

Same philosophy as job_store: the filesystem is the database, every write is
atomic, a reader never sees a half-written session. Flat files (not a directory
per session) because a session has exactly one document and no artifacts.
"""
import time
import uuid
from pathlib import Path

from job_store import atomic_write_json, read_json

TITLE_LEN = 60


class ChatNotFound(Exception):
    pass


class ChatStore:
    def __init__(self, root):
        self.root = Path(root)
        self.root.mkdir(parents=True, exist_ok=True)

    def _path(self, session_id):
        return self.root / (session_id + ".json")

    def create(self, model, now=None):
        now = time.time() if now is None else now
        session = {"session_id": "c_" + uuid.uuid4().hex[:8], "title": "",
                   "model": model, "created_at": now, "updated_at": now,
                   "messages": []}
        atomic_write_json(self._path(session["session_id"]), session)
        return session

    def get(self, session_id):
        try:
            return read_json(self._path(session_id))
        except FileNotFoundError:
            raise ChatNotFound(session_id)

    def _save(self, session, now):
        session["updated_at"] = time.time() if now is None else now
        atomic_write_json(self._path(session["session_id"]), session)
        return session

    def append(self, session_id, message, now=None):
        session = self.get(session_id)
        session["messages"].append(message)
        if not session["title"] and message.get("role") == "user":
            session["title"] = (message.get("content") or "")[:TITLE_LEN]
        return self._save(session, now)

    def set_model(self, session_id, model, now=None):
        session = self.get(session_id)
        session["model"] = model
        return self._save(session, now)

    def delete(self, session_id):
        try:
            self._path(session_id).unlink()
        except FileNotFoundError:
            raise ChatNotFound(session_id)

    def list(self):
        out = []
        for p in sorted(self.root.glob("c_*.json")):
            s = read_json(p)
            out.append({"session_id": s["session_id"], "title": s["title"],
                        "model": s["model"], "updated_at": s["updated_at"],
                        "messages": len(s["messages"])})
        out.sort(key=lambda s: s["updated_at"], reverse=True)
        return out
```

```python
    # ---- chat ----

    def chat_list(self):
        return {"sessions": self.chats.list()}

    def chat_create(self, model):
        self._check_model(model)
        return self.chats.create(model)

    def chat_get(self, session_id):
        return self.chats.get(session_id)

    def chat_delete(self, session_id):
        self.chats.delete(session_id)
        return {"session_id": session_id, "deleted": True}

    def chat_set_model(self, session_id, model):
        self._check_model(model)
        return self.chats.set_model(session_id, model)

    def chat_reply(self, session_id, content):
        """Generator: ("start", None), then ("chunk", text)..., then one
        terminal ("done", metrics) or ("error", message).

        Everything that can refuse -- unknown session, blank message, busy
        GPU -- raises before the first yield, while the web layer can still
        answer a status code. The GPU lock is held from before the first
        token to after the last: a job must not swap VRAM under us.
        """
        session = self.chats.get(session_id)
        if not isinstance(content, str) or not content.strip():
            raise ValueError("empty message")
        lock = self._gpu_guard()
        finished = False
        parts = []
        try:
            self.chats.append(session_id, {"role": "user", "content": content,
                                           "ts": time.time()})
            yield "start", None
            history = [{"role": m["role"], "content": m["content"]}
                       for m in self.chats.get(session_id)["messages"]]
            stream = ollama_client.chat_stream(session["model"], history)
            while True:
                try:
                    text = next(stream)
                except StopIteration as stop:
                    metrics = stop.value or {}
                    break
                except Exception as e:
                    self.chats.append(session_id, {
                        "role": "assistant", "content": "".join(parts),
                        "ts": time.time(), "error": str(e)})
                    finished = True
                    yield "error", str(e)
                    return
                parts.append(text)
                yield "chunk", text
            self.chats.append(session_id, {
                "role": "assistant", "content": metrics.get("content", "".join(parts)),
                "ts": time.time(),
                "metrics": {k: metrics.get(k) for k in
                            ("tokens_per_s", "ttft_s", "eval_count")}})
            finished = True
            yield "done", {"tokens_per_s": metrics.get("tokens_per_s"),
                           "ttft_s": metrics.get("ttft_s"),
                           "eval_count": metrics.get("eval_count")}
        finally:
            # Client gone mid-stream (GeneratorExit lands here): keep the partial.
            if not finished and parts:
                self.chats.append(session_id, {
                    "role": "assistant", "content": "".join(parts),
                    "ts": time.time(), "error": "interrupted"})
            lock.release()
```

```javascript
async function sendChat(sessionId, content, bubble, meta) {
  const resp = await fetch("/api/chats/" + sessionId + "/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json", "X-Factory-Token": TOKEN },
    body: JSON.stringify({ content }),
  });
  if (!resp.ok) {
    const data = await resp.json();
    throw new Error(data.error + ": " + data.detail);
  }
  const reader = resp.body.getReader();
  const dec = new TextDecoder();
  let buf = "";
  const feed = (line) => {
    if (!line.trim()) return;
    const msg = JSON.parse(line);
    if (msg.chunk) bubble.textContent += msg.chunk;
    else if (msg.done) meta.textContent =
      Math.round(msg.tokens_per_s || 0) + " tok/s · ttft " +
      (msg.ttft_s || 0).toFixed(1) + " s";
    else if (msg.error) { meta.textContent = "erreur : " + msg.error; meta.classList.add("chat-error"); }
  };
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    buf += dec.decode(value, { stream: true });
    const lines = buf.split("\n");
    buf = lines.pop();
    lines.forEach(feed);
  }
  feed(buf);
}
```
